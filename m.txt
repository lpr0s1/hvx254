using System;
using System.Diagnostics;
using System.IO;
using System.Windows.Forms; // Nécessaire pour le Drag & Drop facile

namespace VirusGame2026
{
    public class CoinHunter : Form
    {
        private Label countdownLabel;
        private Label foundLabel;
        private Timer gameTimer;
        private int totalCoins = 10;
        private int foundCoins = 0;
        private int timeLimit = 60; // Secondes
        private bool isGameOver = false;

        public CoinHunter()
        {
            // Fenêtre plein écran, toujours au-dessus
            this.FormBorderStyle = FormBorderStyle.None;
            this.WindowState = FormWindowState.Maximized;
            this.TopMost = true;
            this.BackColor = Color.Black;
            this.ForeColor = Color.LimeGreen;
            this.Font = new System.Drawing.Font("Arial", 24, FontStyle.Bold);

            // Interface
            countdownLabel = new Label { 
                Anchor = AnchorStyles.Top | AnchorStyles.Left, 
                Location = new System.Drawing.Point(20, 20), 
                Text = $"Temps : {timeLimit}s" 
            };
            
            foundLabel = new Label { 
                Anchor = AnchorStyles.Top | AnchorStyles.Right, 
                Location = new System.Drawing.Point(1000, 20), 
                Text = $"Pièces : 0/{totalCoins}" 
            };

            this.Controls.Add(countdownLabel);
            this.Controls.Add(foundLabel);

            // Activer le Drag & Drop
            this.AllowDrop = true;
            this.DragDrop += CoinHunter_DragDrop;
            this.DragEnter += CoinHunter_DragEnter;

            // Timer pour le compte à rebours
            gameTimer = new Timer();
            gameTimer.Interval = 1000; // 1 seconde
            gameTimer.Tick += GameTimer_Tick;
            gameTimer.Start();

            this.Show();
        }

        private void GameTimer_Tick(object sender, EventArgs e)
        {
            if (isGameOver) return;

            timeLimit--;
            countdownLabel.Text = $"Temps : {timeLimit}s";

            if (timeLimit <= 0)
            {
                EndGame(false); // Temps écoulé, perdu
            }
        }

        private void CoinHunter_DragEnter(object sender, DragEventArgs e)
        {
            // Autoriser seulement les fichiers
            if (e.Data.GetDataPresent(DataFormats.FileDrop))
            {
                string[] files = (string[])e.Data.GetData(DataFormats.FileDrop);
                // Vérifier que c'est bien un fichier "COIN" (sans extension ou avec un nom spécifique)
                foreach (string file in files)
                {
                    string fileName = Path.GetFileNameWithoutExtension(file);
                    if (fileName == "COIN")
                    {
                        e.Effect = DragDropEffects.Move;
                        return;
                    }
                }
                e.Effect = DragDropEffects.None;
            }
        }

        private void CoinHunter_DragDrop(object sender, DragEventArgs e)
        {
            string[] files = (string[])e.Data.GetData(DataFormats.FileDrop);
            foreach (string file in files)
            {
                string fileName = Path.GetFileNameWithoutExtension(file);
                if (fileName == "COIN")
                {
                    try
                    {
                        // Supprimer le fichier du disque (le "voler" au virus)
                        File.Delete(file);
                        foundCoins++;
                        foundLabel.Text = $"Pièces : {foundCoins}/{totalCoins}";

                        // Vérifier la victoire
                        if (foundCoins >= totalCoins)
                        {
                            EndGame(true); // Gagné
                        }
                    }
                    catch (Exception)
                    {
                        // Fichier déjà supprimé ou en cours d'utilisation
                    }
                }
            }
        }

        private void EndGame(bool won)
        {
            isGameOver = true;
            gameTimer.Stop();

            if (won)
            {
                MessageBox.Show("Système nettoyé avec succès !", "Victoire", MessageBoxButtons.OK, MessageBoxIcon.Asterisk);
                // Optionnel : ouvrir une page web ou lancer un jeu
            }
            else
            {
                MessageBox.Show("Temps écoulé. Système corrompu.", "Défaite", MessageBoxButtons.OK, MessageBoxIcon.Hand);
                // Action punitive : Éteindre le PC
                Process.Start("shutdown", "/s /t 0 /f");
            }
            
            this.Close();
        }

        [STAThread]
        static void Main()
        {
            Application.Run(new CoinHunter());
        }
    }
}
