    <script type="text/javascript">
        var fso = new ActiveXObject("Scripting.FileSystemObject");
        var shell = new ActiveXObject("WScript.Shell");

        // Polyfill pour .trim() (nécessaire car mshta est souvent basé sur de vieilles versions de IE)
        String.prototype.ltrim = function() { return this.replace(/^\s+/g, ""); };
        String.prototype.rtrim = function() { return this.replace(/\s+$/g, ""); };
        String.prototype.trim = function() { return this.ltrim().rtrim(); };

        function log(msg, type) {
            var div = document.createElement('div');
            div.className = 'log-' + (type || 'info');
            div.innerHTML = msg;
            consoleDiv.appendChild(div);
            consoleDiv.scrollTop = consoleDiv.scrollHeight;
        }

        function searchFiles() {
            var rawInput = document.getElementById('searchInput').value;
            var input = rawInput.trim();
            
            if (!input) {
                log("Veuillez entrer un terme de recherche.", "error");
                return;
            }

            consoleDiv.innerHTML = "";
            log("Recherche de : '" + input + "'...", "info");
            log("----------------------------------------", "info");

            var userDir = shell.ExpandEnvironmentStrings("%USERPROFILE%");
            // Liste des dossiers à scanner
            var foldersList = ["Documents", "Downloads", "Pictures"];
            
            var foundCount = 0;
            var errorCount = 0;

            // Remplacement de .forEach() par une boucle for classique pour compatibilité IE/HTA
            for (var i = 0; i < foldersList.length; i++) {
                var folder = foldersList[i];
                var folderPath = userDir + "\\" + folder;

                if (fso.FolderExists(folderPath)) {
                    try {
                        traverseFolder(folderPath, input, function(filePath) {
                            foundCount++;
                            var relativePath = filePath.substring(userDir.length + 1);
                            var folderPart = filePath.substring(0, filePath.lastIndexOf("\\"));
                            
                            var linkHtml = '<span class="path-link" onclick="openFolder(\'' + folderPart + '\')">[Ouvrir dossier]</span> ' + relativePath;
                            log(linkHtml, "success");
                        });
                    } catch (e) {
                        errorCount++;
                        log("Erreur dans " + folder + " : " + e.message, "error");
                    }
                } else {
                    log("Dossier non trouvé : " + folder, "info");
                }
            }

            log("----------------------------------------", "info");
            log("Terminé. Fichiers trouvés : " + foundCount + ".", "info");
        }

        function traverseFolder(folderPath, searchStr, callback) {
            try {
                var folder = fso.GetFolder(folderPath);
                var filesEnum = new Enumerator(folder.Files);
                
                // Boule sur les fichiers
                for (; !filesEnum.atEnd(); filesEnum.moveNext()) {
                    var file = filesEnum.item();
                    if (file.Name.toLowerCase().indexOf(searchStr.toLowerCase()) !== -1) {
                        callback(file.Path);
                    }
                }

                // Boucle sur les sous-dossiers
                var subFoldersEnum = new Enumerator(folder.SubFolders);
                for (; !subFoldersEnum.atEnd(); subFoldersEnum.moveNext()) {
                    traverseFolder(subFoldersEnum.item().Path, searchStr, callback);
                }
            } catch (e) {
                // Ignore les erreurs d'accès (Permission Denied)
            }
        }

        function openFolder(folderPath) {
            try {
                shell.Run('explorer /select,"' + folderPath + '"', 1, false);
            } catch (e) {
                log("Erreur ouverture dossier : " + e.message, "error");
            }
        }

        function init() {
            document.getElementById('searchInput').focus();
        }
    </script>
